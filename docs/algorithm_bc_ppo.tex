\begin{algorithm}[t]
\caption{Expert collection and behavior cloning (BC)}
\label{alg:bc}
\begin{algorithmic}[1]
\Require SUMO road with background traffic; IDM vehicles as experts; per-route quotas $q_{\mathrm{tr}}, q_{\mathrm{val}}$
\Ensure BC policy $\pi_{\mathrm{BC}}$ that takes raw observations
\Statex \textit{Expert collection}
\Repeat
  \State Run a fresh 3{,}600\,s simulation; assign the run to validation with probability $1/20$, else to training
  \For{each closed IDM trajectory and each step $t$}
    \State $o_t \gets$ the 52-dim observation computed for that vehicle as if it were the ego
    \State $a^{\mathrm{acc}}_t \gets (v_{t+1}-v_t)/(\Delta t\, a_{\max})$,\quad $y_t \gets$ SUMO lane change $\in\{+1,0,-1\}$
    \State Keep $(o_t, a^{\mathrm{acc}}_t, y_t)$ if $|a^{\mathrm{acc}}_t|\le 1$ and $v_{t+1}\le v_{\mathrm{lim}}$
  \EndFor
\Until{every route has $q_{\mathrm{tr}}$ training and $q_{\mathrm{val}}$ validation trajectories}
\State Select whole trajectories per route, balanced over three entry-time bands $\Rightarrow \Dtr, \Dval$
\Statex \textit{Behavior cloning}
\State $m_x, \sigma_x \gets$ per-feature mean and std of $o$ over $\Dtr$
\State $w_c \gets (N/(3 n_c))^{1/r}$ for $c\in\{\text{left},\text{keep},\text{right}\}$; normalize so that $\tfrac13\sum_c w_c = 1$
\State Initialize actor $\pi_\theta$: MLP $52\!\to\!256\!\to\!256$ (ReLU), heads $\mu_\theta=\tanh(\cdot)$ and lane logits $\ell_\theta\in\mathbb{R}^3$
\For{epoch $=1,\dots,E$}
  \For{each minibatch $\mathcal{B}\subset\Dtr$ of size $B$}
    \State $\displaystyle \mathcal{L}_{\mathrm{BC}} = \frac{1}{|\mathcal{B}|}\sum_{i\in\mathcal{B}}\big(\mu_\theta(\bar o_i)-a^{\mathrm{acc}}_i\big)^2
      + \frac{\sum_{i} w_{y_i}\,\CE(\ell_\theta(\bar o_i), y_i)}{\sum_{i} w_{y_i}}$, \quad $\bar o = (o-m_x)/\sigma_x$
    \State $\theta \gets \mathrm{Adam}(\theta, \nabla_\theta \mathcal{L}_{\mathrm{BC}})$ \Comment{gradient norm clipped to 1}
  \EndFor
  \State Evaluate $\mathcal{L}_{\mathrm{BC}}$ on $\Dval$; keep $\theta_{\mathrm{BC}}$ with the lowest value
\EndFor
\State Fold normalization into the first layer: $W_1 \gets W_1\,\mathrm{diag}(1/\sigma_x)$, $b_1 \gets b_1 - W_1 m_x$
\State \Return $\pi_{\mathrm{BC}} = \pi_{\theta_{\mathrm{BC}}}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Anchored PPO fine-tuning from the BC policy}
\label{alg:ppo}
\begin{algorithmic}[1]
\Require $\pi_{\mathrm{BC}}$; $N$ parallel environments; event reward $r_t$; hyperparameters in Table~\ref{tab:hparams}
\Ensure Fine-tuned policy $\pi_{\theta^\star}$
\State $\theta \gets \theta_{\mathrm{BC}}$; scale the lane head by $\kappa$: $\ell_\theta \gets \kappa\,\ell_\theta$ \Comment{sharper lane sampling, same argmax}
\State Acceleration policy $\mathcal{N}(\mu_\theta(s), \sigma^2)$ with learnable $\sigma$, $\sigma \gets \sigma_0$
\State Frozen anchor $\pi_{\mathrm{anc}} \gets \pi_\theta$ \Comment{sharpened BC policy}
\State Initialize critic $V_\phi$ (separate MLP, inputs standardized by $m_x,\sigma_x$)
\For{iteration $k=1,\dots,K$}
  \State Roll out $\pi_\theta$ for $T$ steps in each of $N$ environments; random route, start lane, and signal phase per episode
  \State Store $(s_t, a_t, \log\pi_{\mathrm{old}}(a_t\mid s_t), r_t, d_t)$, where $a_t = (a^{\mathrm{acc}}_t, a^{\mathrm{lane}}_t)$
  \State $\delta_t = r_t + \gamma(1-d_t)V_\phi(s_{t+1}) - V_\phi(s_t)$,\quad $\hat A_t = \delta_t + \gamma\lambda(1-d_t)\hat A_{t+1}$,\quad $R_t = \hat A_t + V_\phi(s_t)$
  \State Normalize $\hat A$ over the batch to mean 0, std 1
  \For{epoch $=1,\dots,K_V$} \Comment{critic, every iteration}
    \For{each minibatch $\mathcal{M}$} \State $\phi \gets \mathrm{Adam}\big(\phi, \nabla_\phi \tfrac{1}{|\mathcal{M}|}\sum (V_\phi(s)-R)^2\big)$ \EndFor
  \EndFor
  \If{$k > K_{\mathrm{warm}}$} \Comment{actor after critic warm-up}
    \For{epoch $=1,\dots,K_\pi$}
      \For{each minibatch $\mathcal{M}$}
        \State $\rho = \pi_\theta(a\mid s)/\pi_{\mathrm{old}}(a\mid s)$, with $\log\pi = \log\mathcal{N}(a^{\mathrm{acc}};\mu_\theta,\sigma^2) + \log\mathrm{softmax}(\ell_\theta)_{a^{\mathrm{lane}}}$
        \If{$\frac{1}{|\mathcal{M}|}\sum(\rho - 1 - \log\rho) > 1.5\,\delta_{\mathrm{KL}}$} \textbf{stop} the actor update of this iteration \EndIf
        \State $\displaystyle \mathcal{L}_{\mathrm{clip}} = -\frac{1}{|\mathcal{M}|}\sum \min\!\big(\rho\hat A,\ \mathrm{clip}(\rho,1-\epsilon,1+\epsilon)\hat A\big)$
        \State $\displaystyle \mathcal{L}_{\mathrm{anc}} = \frac{1}{|\mathcal{M}|}\sum\Big[\frac{(\mu_\theta-\mu_{\mathrm{anc}})^2}{2\sigma_0^2} + \KL\big(\pi^{\mathrm{lane}}_\theta\,\|\,\pi^{\mathrm{lane}}_{\mathrm{anc}}\big)\Big]$
        \State $\mathcal{L}_\pi = \mathcal{L}_{\mathrm{clip}} + \beta\,\mathcal{L}_{\mathrm{anc}}$ \Comment{no entropy term}
        \State $\theta \gets \mathrm{Adam}(\theta, \nabla_\theta \mathcal{L}_\pi)$ \Comment{gradient norm clipped to 0.5}
      \EndFor
    \EndFor
  \EndIf
\EndFor
\State $\theta^\star \gets$ the saved checkpoint with the highest success rate on the validation suite (later on ties)
\State \Return $\pi_{\theta^\star}$ \Comment{evaluated deterministically: $a^{\mathrm{acc}} = \mu$, $a^{\mathrm{lane}} = \arg\max \ell$}
\end{algorithmic}
\end{algorithm}
